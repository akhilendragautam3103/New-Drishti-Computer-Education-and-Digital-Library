<script>

/* ================= PASSAGES ================= */

const passages = [

`Computer education is very important in the modern world. Every student should learn basic computer skills because computers are used in schools, offices, banks, shops and many other places. Regular typing practice helps students improve their speed and accuracy.`,

`Typing is an important computer skill that can save time and increase productivity. A good typist should focus on accuracy first and speed later. Proper finger placement, correct posture and regular practice can help a student become a confident and efficient typist.`,

`Technology has changed the way people communicate, learn and work. Computers and the internet provide many opportunities for education and professional development. Students who practice typing regularly can complete their assignments faster and can develop better keyboard control and concentration.`
];


/* ================= VARIABLES ================= */

let currentText = "";

let timerSeconds = 300;

let remainingSeconds = 300;

let timerInterval = null;

let isRunning = false;

let isPaused = false;

let startTime = null;


/* ================= ELEMENTS ================= */

const textDisplay =
    document.getElementById("textDisplay");

const typingInput =
    document.getElementById("typingInput");

const timer =
    document.getElementById("timer");

const wpm =
    document.getElementById("wpm");

const cpm =
    document.getElementById("cpm");

const accuracy =
    document.getElementById("accuracy");

const correct =
    document.getElementById("correct");

const errors =
    document.getElementById("errors");

const progress =
    document.getElementById("progress");

const startBtn =
    document.getElementById("startBtn");

const pauseBtn =
    document.getElementById("pauseBtn");

const restartBtn =
    document.getElementById("restartBtn");

const timeSelect =
    document.getElementById("timeSelect");

const passageSelect =
    document.getElementById("passageSelect");

const resultBox =
    document.getElementById("resultBox");


/* ================= LOAD PASSAGE ================= */

function loadPassage(){

    currentText =
        passages[
            Number(passageSelect.value)
        ];

    textDisplay.innerHTML = "";

    currentText.split("").forEach(
        function(char,index){

            const span =
                document.createElement("span");

            span.textContent = char;

            span.id =
                "char-" + index;

            textDisplay.appendChild(span);

        }
    );

}


/* ================= FORMAT TIME ================= */

function formatTime(seconds){

    const minutes =
        Math.floor(seconds / 60);

    const secs =
        seconds % 60;

    return String(minutes).padStart(2,"0")
        + ":" +
        String(secs).padStart(2,"0");

}


/* ================= RESET ================= */

function resetTest(){

    clearInterval(timerInterval);

    isRunning = false;

    isPaused = false;

    timerSeconds =
        Number(timeSelect.value);

    remainingSeconds =
        timerSeconds;

    timer.textContent =
        formatTime(remainingSeconds);

    typingInput.value = "";

    typingInput.disabled = true;

    wpm.textContent = "0";

    cpm.textContent = "0";

    accuracy.textContent = "100%";

    correct.textContent = "0";

    errors.textContent = "0";

    progress.style.width = "0%";

    resultBox.style.display = "none";

    startBtn.disabled = false;

    pauseBtn.disabled = false;

    loadPassage();

}


/* ================= START TEST ================= */

function startTest(){

    if(isRunning){
        return;
    }

    isRunning = true;

    isPaused = false;

    typingInput.disabled = false;

    typingInput.focus();

    startTime = Date.now();

    clearInterval(timerInterval);

    timerInterval =
        setInterval(function(){

            if(!isPaused){

                remainingSeconds--;

                timer.textContent =
                    formatTime(remainingSeconds);

                updateStats();

                if(remainingSeconds <= 0){

                    finishTest();

                }

            }

        },1000);

}


/* ================= PAUSE ================= */

function pauseTest(){

    if(!isRunning){
        return;
    }

    isPaused = !isPaused;

    if(isPaused){

        pauseBtn.textContent =
            "▶ Resume";

        typingInput.blur();

    }else{

        pauseBtn.textContent =
            "⏸ Pause";

        typingInput.focus();

    }

}


/* ================= TYPING EVENT ================= */

typingInput.addEventListener(
    "input",
    function(){

        if(!isRunning || isPaused){
            return;
        }

        updateCharacterDisplay();

        updateStats();

    }
);


/* ================= CHARACTER CHECK ================= */

function updateCharacterDisplay(){

    const typedText =
        typingInput.value;

    const spans =
        textDisplay.querySelectorAll("span");

    spans.forEach(function(span,index){

        span.className = "";

        if(index < typedText.length){

            if(
                typedText[index]
                ===
                currentText[index]
            ){

                span.classList.add(
                    "correct"
                );

            }else{

                span.classList.add(
                    "incorrect"
                );

            }

        }

        else if(index === typedText.length){

            span.classList.add(
                "current"
            );

        }

    });

}


/* ================= CALCULATE STATS ================= */

function calculateStats(){

    const typed =
        typingInput.value;

    let correctChars = 0;

    let errorChars = 0;


    for(
        let i = 0;
        i < typed.length;
        i++
    ){

        if(
            typed[i]
            ===
            currentText[i]
        ){

            correctChars++;

        }else{

            errorChars++;

        }

    }


    const totalTyped =
        typed.length;


    const elapsedSeconds =
        Math.max(
            1,
            timerSeconds - remainingSeconds
        );


    const elapsedMinutes =
        elapsedSeconds / 60;


    /*
       Standard Typing Formula:

       WPM =
       Correct characters / 5 / minutes
    */

    const calculatedWPM =
        correctChars > 0
        ?
        Math.round(
            (correctChars / 5)
            / elapsedMinutes
        )
        :
        0;


    /*
       CPM =
       Correct characters / minutes
    */

    const calculatedCPM =
        correctChars > 0
        ?
        Math.round(
            correctChars
            / elapsedMinutes
        )
        :
        0;


    /*
       Accuracy =
       Correct / Total Typed × 100
    */

    const calculatedAccuracy =
        totalTyped > 0
        ?
        Math.round(
            (correctChars / totalTyped) * 100
        )
        :
        100;


    return {

        correct:correctChars,

        errors:errorChars,

        total:totalTyped,

        wpm:calculatedWPM,

        cpm:calculatedCPM,

        accuracy:calculatedAccuracy

    };

}


/* ================= UPDATE LIVE STATS ================= */

function updateStats(){

    const stats =
        calculateStats();


    wpm.textContent =
        stats.wpm;


    cpm.textContent =
        stats.cpm;


    accuracy.textContent =
        stats.accuracy + "%";


    correct.textContent =
        stats.correct;


    errors.textContent =
        stats.errors;


    /*
       Progress based on
       typed characters
    */

    const percent =
        currentText.length > 0
        ?
        Math.min(
            100,
            (stats.total / currentText.length) * 100
        )
        :
        0;


    progress.style.width =
        percent + "%";


    /*
       Automatically finish
       if passage is completed
    */

    if(
        typingInput.value.length
        >=
        currentText.length
    ){

        finishTest();

    }

}


/* ================= FINISH TEST ================= */

function finishTest(){

    if(!isRunning){
        return;
    }

    clearInterval(timerInterval);

    isRunning = false;

    typingInput.disabled = true;

    const stats =
        calculateStats();


    document.getElementById(
        "finalWpm"
    ).textContent =
        stats.wpm;


    document.getElementById(
        "finalCpm"
    ).textContent =
        stats.cpm;


    document.getElementById(
        "finalAccuracy"
    ).textContent =
        stats.accuracy + "%";


    document.getElementById(
        "finalErrors"
    ).textContent =
        stats.errors;


    let message = "";


    if(stats.wpm >= 50){

        message =
            "🏆 Excellent! Your typing speed is very good.";

    }

    else if(stats.wpm >= 35){

        message =
            "👏 Very Good! Keep practicing to improve further.";

    }

    else if(stats.wpm >= 25){

        message =
            "👍 Good! Regular practice will increase your speed.";

    }

    else{

        message =
            "💪 Keep Practicing! Accuracy first, speed will follow.";

    }


    document.getElementById(
        "resultMessage"
    ).textContent =
        message;


    resultBox.style.display =
        "block";


    resultBox.scrollIntoView({
        behavior:"smooth"
    });

}


/* ================= BUTTON EVENTS ================= */

startBtn.addEventListener(
    "click",
    startTest
);


pauseBtn.addEventListener(
    "click",
    pauseTest
);


restartBtn.addEventListener(
    "click",
    resetTest
);


/* ================= TIME CHANGE ================= */

timeSelect.addEventListener(
    "change",
    resetTest
);


/* ================= PASSAGE CHANGE ================= */

passageSelect.addEventListener(
    "change",
    resetTest
);


/* ================= INITIAL LOAD ================= */

resetTest();

</script>
