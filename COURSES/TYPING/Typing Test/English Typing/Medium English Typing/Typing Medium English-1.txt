/* ================= LOAD PASSAGE ================= */

function loadPassage(){

    let language =
        languageSelect.value;

    let passages =
        language === "hindi"
        ?
        hindiPassages
        :
        englishPassages;


    /*
       Random passage
    */

    const random =
        Math.floor(
            Math.random() * passages.length
        );


    currentText =
        passages[random];


    textDisplay.innerHTML = "";


    if(language === "hindi"){

        textDisplay.classList.add(
            "hindi-text"
        );

        typingInput.classList.add(
            "hindi-input"
        );

    }else{

        textDisplay.classList.remove(
            "hindi-text"
        );

        typingInput.classList.remove(
            "hindi-input"
        );

    }


    currentText.split("").forEach(
        function(char,index){

            const span =
                document.createElement("span");

            span.textContent = char;

            span.id =
                "char-" + index;

            textDisplay.appendChild(span);

        }
    );

}


/* ================= FORMAT TIME ================= */

function formatTime(seconds){

    const minutes =
        Math.floor(seconds / 60);

    const secs =
        seconds % 60;

    return String(minutes).padStart(2,"0")
        + ":" +
        String(secs).padStart(2,"0");

}


/* ================= RESET ================= */

function resetTest(){

    clearInterval(timerInterval);

    isRunning = false;

    isPaused = false;


    timerSeconds =
        Number(timeSelect.value);

    remainingSeconds =
        timerSeconds;


    timer.textContent =
        formatTime(remainingSeconds);


    typingInput.value = "";

    typingInput.disabled = true;


    wpm.textContent = "0";

    cpm.textContent = "0";

    accuracy.textContent = "100%";

    correct.textContent = "0";

    errors.textContent = "0";


    targetDisplay.textContent =
        targetSelect.value;


    progress.style.width = "0%";

    progressText.textContent = "0%";


    resultBox.style.display = "none";


    document.getElementById(
        "pauseBtn"
    ).textContent =
        "⏸ Pause";


    loadPassage();

}


/* ================= START ================= */

function startTest(){

    if(isRunning){
        return;
    }


    if(
        studentName.value.trim() === ""
    ){

        alert(
            "Please enter Student Name."
        );

        studentName.focus();

        return;

    }


    if(
        rollNumber.value.trim() === ""
    ){

        alert(
            "Please enter Roll Number."
        );

        rollNumber.focus();

        return;

    }


    isRunning = true;

    isPaused = false;

    typingInput.disabled = false;

    typingInput.focus();


    clearInterval(timerInterval);


    timerInterval =
    setInterval(function(){

        if(!isPaused){

            remainingSeconds--;

            timer.textContent =
                formatTime(
                    remainingSeconds
                );


            updateStats();


            if(
                remainingSeconds <= 0
            ){

                finishTest();

            }

        }

    },1000);

}


/* ================= PAUSE ================= */

function pauseTest(){

    if(!isRunning){
        return;
    }


    isPaused =
        !isPaused;


    if(isPaused){

        document.getElementById(
            "pauseBtn"
        ).textContent =
            "▶ Resume";

        typingInput.blur();

    }else{

        document.getElementById(
            "pauseBtn"
        ).textContent =
            "⏸ Pause";

        typingInput.focus();

    }

}


/* ================= TYPING ================= */

    function(){

        if(
            !isRunning ||
            isPaused
        ){

            return;

        }


        updateCharacterDisplay();

        updateStats();

    }
);


/* ================= CHARACTER DISPLAY ================= */

function updateCharacterDisplay(){

    const typedText =
        typingInput.value;

    const spans =
        textDisplay.querySelectorAll("span");


    spans.forEach(
        function(span,index){

            span.className = "";


            if(
                index < typedText.length
            ){

                if(
                    typedText[index]
                    ===
                    currentText[index]
                ){

                    span.classList.add(
                        "correct"
                    );

                }else{

                    span.classList.add(
                        "incorrect"
                    );

                }

            }
            else if(
                index === typedText.length
            ){

                span.classList.add(
                    "current"
                );

            }

        }
    );

}


/* ================= CALCULATE ================= */

function calculateStats(){

    const typed =
        typingInput.value;


    let correctChars = 0;

    let errorChars = 0;


    for(
        let i=0;
        i<typed.length;
        i++
    ){

        if(
            typed[i]
            ===
            currentText[i]
        ){

            correctChars++;

        }else{

            errorChars++;

        }

    }


    const totalTyped =
        typed.length;


    const elapsedSeconds =
        Math.max(
            1,
            timerSeconds -
            remainingSeconds
        );


    const elapsedMinutes =
        elapsedSeconds / 60;


    /*
       Standard WPM
       5 characters = 1 word
    */

    const calculatedWPM =
        correctChars > 0
        ?
        Math.round(
            (correctChars / 5)
            /
            elapsedMinutes
        )
        :
        0;


    const calculatedCPM =
        correctChars > 0
        ?
        Math.round(
            correctChars
            /
            elapsedMinutes
        )
        :
        0;


    const calculatedAccuracy =
        totalTyped > 0
        ?
        Math.round(
            (
                correctChars /
                totalTyped
            ) * 100
        )
        :
        100;


    return{

        correct:correctChars,

        errors:errorChars,

        total:totalTyped,

        wpm:calculatedWPM,

        cpm:calculatedCPM,

        accuracy:calculatedAccuracy

    };

}


/* ================= UPDATE STATS ================= */

function updateStats(){

    const stats =
        calculateStats();


    wpm.textContent =
        stats.wpm;


    cpm.textContent =
        stats.cpm;


    accuracy.textContent =
        stats.accuracy + "%";


    correct.textContent =
        stats.correct;


    errors.textContent =
        stats.errors;


    const percent =
        currentText.length > 0
        ?
        Math.min(
            100,
            (
                stats.total /
                currentText.length
            ) * 100
        )
        :
        0;


    progress.style.width =
        percent + "%";


    progressText.textContent =
        Math.round(percent) + "%";


    /*
       Complete passage
    */

    if(
        typingInput.value.length
        >=
        currentText.length
    ){

        finishTest();

    }

}


/* ================= FINISH ================= */

function finishTest(){

    if(!isRunning){
        return;
    }


    clearInterval(timerInterval);

    isRunning = false;

    typingInput.disabled = true;


    const stats =
        calculateStats();


    /* Student */

    document.getElementById(
        "resultName"
    ).textContent =
        studentName.value;


    document.getElementById(
        "resultRoll"
    ).textContent =
        rollNumber.value;


    /* Result */

    document.getElementById(
        "finalWpm"
    ).textContent =
        stats.wpm;


    document.getElementById(
        "finalCpm"
    ).textContent =
        stats.cpm;


    document.getElementById(
        "finalAccuracy"
    ).textContent =
        stats.accuracy + "%";


    document.getElementById(
        "finalErrors"
    ).textContent =
        stats.errors;


    /* Target */

    const target =
        Number(
            targetSelect.value
        );


    document.getElementById(
        "resultTarget"
    ).textContent =
        target + " WPM";


    /* Language */

    const language =
        languageSelect.value ===
        "hindi"
        ?
        "Hindi"
        :
        "English";


    document.getElementById(
        "resultLanguage"
    ).textContent =
        language;


    /* Message */

    let message = "";


    if(
        stats.wpm >= target &&
        stats.accuracy >= 95
    ){

        message =
        "🏆 Excellent! Target achieved with excellent accuracy.";

    }

    else if(
        stats.wpm >= target
    ){

        message =
        "🎉 Target WPM achieved! Improve accuracy further.";

    }

    else if(
        stats.wpm >= target - 5
    ){

        message =
        "👏 Very close! Keep practicing to achieve the target.";

    }

    else if(
        stats.wpm >= 25
    ){

        message =
        "👍 Good performance. Regular practice will improve your speed.";

    }

    else{

        message =
        "💪 Keep practicing. Focus on accuracy first.";

    }


    document.getElementById(
        "resultMessage"
    ).textContent =
        message;


    resultBox.style.display =
        "block";


    resultBox.scrollIntoView({
        behavior:"smooth"
    });

}
